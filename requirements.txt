# ---------------------------------------------------------------------------
# Runtime dependencies.
#
# Deliberately small. Every entry below is justified in docs/DESIGN.md; the
# short version is inline so the list can be audited without leaving the file.
# ---------------------------------------------------------------------------

# Web layer. uvicorn[standard] pulls uvloop/httptools where the platform
# supports them, which is where most of the "free" async throughput comes from.
fastapi==0.121.2
uvicorn[standard]==0.38.0
pydantic==2.12.4
pydantic-settings==2.13.1

# Storage. SQLite is the durable source of truth (stdlib driver); aiosqlite
# wraps it in a thread executor so writes never block the event loop.
aiosqlite==0.21.0

# Vector maths. numpy alone is enough: at 50k docs an exact brute-force
# dot-product top-k runs in ~2.5ms, which is cheaper than the query embedding
# that feeds it. No ANN library earns its complexity at this scale.
numpy==2.3.5

# Embeddings. fastembed runs sentence-transformer models on onnxruntime, which
# keeps the image around ~400MB instead of the ~2.5GB a torch install costs and
# cuts cold-start time — this matters because the brief calls out frequent
# restarts.
fastembed==0.8.0

# onnxruntime is pinned below 1.20 on purpose. Builds from 1.20 onward assume
# newer CPU instruction sets and abort with SIGILL on older cores, which is
# exactly what free-tier and shared cloud vCPUs tend to be. 1.19.2 targets a
# wider baseline. See docs/DESIGN.md ("Embedding runtime") for the full trace.
onnxruntime==1.19.2
